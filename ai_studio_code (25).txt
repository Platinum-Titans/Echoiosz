public class ExtensionEngine: NSObject, ObservableObject {
    public static let shared = ExtensionEngine()
    
    @Published public var installedExtensions: [EchoExtension] = []
    private var providers: [String: ExtensionProvider] = [:]
    
    override private init() {
        super.init()
        registerProvider(MockLocalExtensionProvider())
    }
    
    public func registerProvider(_ provider: ExtensionProvider) {
        let manifest = provider.manifest
        providers[manifest.id] = provider
        if !installedExtensions.contains(where: { $0.id == manifest.id }) {
            installedExtensions.append(manifest)
        }
    }
    
    public func toggleExtension(id: String) {
        if let idx = installedExtensions.firstIndex(where: { $0.id == id }) {
            installedExtensions[idx].enabled.toggle()
        }
    }
    
    public func searchAcrossProviders(query: String) async -> [Track] {
        var results: [Track] = []
        for ext in installedExtensions where ext.enabled {
            if let provider = providers[ext.id] {
                if let tracks = try? await provider.searchTracks(query: query) {
                    results.append(contentsOf: tracks)
                }
            }
        }
        return results
    }
    
    public func fetchAllShelves() async -> [Shelf] {
        var shelves: [Shelf] = []
        for ext in installedExtensions where ext.enabled {
            if let provider = providers[ext.id] {
                if let s = try? await provider.fetchShelves() {
                    shelves.append(contentsOf: s)
                }
            }
        }
        return shelves
    }
    
    public func resolveStreamUrl(for track: Track) async -> URL? {
        if let provider = providers[track.extensionId] {
            return try? await provider.resolveStreamUrl(for: track)
        }
        if let first = providers.values.first {
            return try? await first.resolveStreamUrl(for: track)
        }
        return nil
    }
}
