import Foundation
import AVFoundation
import MediaPlayer
import Combine

public enum RepeatMode: String, CaseIterable {
    case off = "Off"
    case all = "All"
    case one = "One"
}

public class AudioPlayerService: NSObject, ObservableObject {
    public static let shared = AudioPlayerService()
    
    @Published public var currentTrack: Track?
    @Published public var isPlaying: Bool = false
    @Published public var isLoading: Bool = false
    @Published public var currentTime: TimeInterval = 0
    @Published public var duration: TimeInterval = 0
    @Published public var queue: [Track] = []
    @Published public var queueIndex: Int = 0
    @Published public var shuffle: Bool = false
    @Published public var repeatMode: RepeatMode = .off
    @Published public var volume: Float = 1.0
    @Published public var isMuted: Bool = false
    @Published public var activeLyricIndex: Int = -1
    
    private var player: AVPlayer?
    private var timeObserverToken: Any?
    private var cancellables = Set<AnyCancellable>()
    
    override private init() {
        super.init()
        setupAudioSession()
        setupNotificationObservers()
    }
    
    deinit {
        removeTimeObserver()
    }
    
    private func setupAudioSession() {
        do {
            let session = AVAudioSession.sharedInstance()
            try session.setCategory(.playback, mode: .default, options: [.allowAirPlay, .allowBluetooth, .allowBluetoothA2DP])
            try session.setActive(true)
        } catch {
            print("[AudioPlayerService] AudioSession setup error: \(error)")
        }
    }
    
    private func setupNotificationObservers() {
        NotificationCenter.default.publisher(for: .AVPlayerItemDidPlayToEndTime)
            .sink { [weak self] _ in
                self?.handleTrackEnded()
            }
            .store(in: &cancellables)
            
        NotificationCenter.default.publisher(for: AVAudioSession.interruptionNotification)
            .sink { [weak self] notification in
                self?.handleAudioInterruption(notification: notification)
            }
            .store(in: &cancellables)
            
        NotificationCenter.default.publisher(for: AVAudioSession.routeChangeNotification)
            .sink { [weak self] notification in
                self?.handleRouteChange(notification: notification)
            }
            .store(in: &cancellables)
    }
    
    public func playTrack(_ track: Track, queue: [Track]? = nil) {
        if let newQueue = queue, !newQueue.isEmpty {
            self.queue = newQueue
            self.queueIndex = newQueue.firstIndex(where: { $0.id == track.id }) ?? 0
        } else if self.queue.isEmpty {
            self.queue = [track]
            self.queueIndex = 0
        }
        
        self.currentTrack = track
        self.duration = track.duration > 0 ? track.duration : 180
        self.currentTime = 0
        self.isLoading = true
        self.activeLyricIndex = -1
        
        LibraryService.shared.recordHistory(track: track)
        
        if let cachedUrl = OfflineCacheManager.shared.getLocalAudioFileURL(for: track.id) {
            playFromUrl(cachedUrl)
            return
        }
        
        if let streamUrlString = track.streamUrl, let remoteUrl = URL(string: streamUrlString) {
            playFromUrl(remoteUrl)
        } else {
            Task {
                if let resolved = await ExtensionEngine.shared.resolveStreamUrl(for: track) {
                    await MainActor.run {
                        self.playFromUrl(resolved)
                    }
                } else {
                    await MainActor.run {
                        self.isLoading = false
                    }
                }
            }
        }
        
        LyricsService.shared.fetchLyrics(for: track) { [weak self] syncedLyrics in
            DispatchQueue.main.async {
                if self?.currentTrack?.id == track.id {
                    self?.currentTrack?.lyrics = syncedLyrics
                }
            }
        }
    }
    
    private func playFromUrl(_ url: URL) {
        removeTimeObserver()
        
        let asset = AVURLAsset(url: url)
        let playerItem = AVPlayerItem(asset: asset)
        
        if player == nil {
            player = AVPlayer(playerItem: playerItem)
            player?.automaticallyWaitsToMinimizeStalling = true
        } else {
            player?.replaceCurrentItem(with: playerItem)
        }
        
        player?.volume = isMuted ? 0 : volume
        player?.play()
        self.isPlaying = true
        self.isLoading = false
        
        addTimeObserver()
        updateNowPlaying()
    }
    
    public func play() {
        guard let player = player else {
            if let track = currentTrack { playTrack(track) }
            return
        }
        player.play()
        self.isPlaying = true
        updateNowPlayingPlaybackRate(1.0)
    }
    
    public func pause() {
        player?.pause()
        self.isPlaying = false
        updateNowPlayingPlaybackRate(0.0)
    }
    
    public func togglePlayPause() {
        if isPlaying { pause() } else { play() }
    }
    
    public func seek(to time: TimeInterval) {
        let cmTime = CMTime(seconds: time, preferredTimescale: 600)
        player?.seek(to: cmTime, toleranceBefore: .zero, toleranceAfter: .zero) { [weak self] _ in
            self?.currentTime = time
            self?.updateLyricPosition(time: time)
            self?.updateNowPlaying()
        }
    }
    
    public func playNext() {
        guard !queue.isEmpty else { return }
        if shuffle {
            queueIndex = Int.random(in: 0..<queue.count)
        } else if queueIndex + 1 < queue.count {
            queueIndex += 1
        } else if repeatMode == .all {
            queueIndex = 0
        } else {
            pause()
            return
        }
        playTrack(queue[queueIndex])
    }
    
    public func playPrevious() {
        guard !queue.isEmpty else { return }
        if currentTime > 3.0 {
            seek(to: 0)
            return
        }
        if queueIndex - 1 >= 0 {
            queueIndex -= 1
        } else if repeatMode == .all {
            queueIndex = queue.count - 1
        }
        playTrack(queue[queueIndex])
    }
    
    public func toggleShuffle() {
        shuffle.toggle()
    }
    
    public func cycleRepeatMode() {
        switch repeatMode {
        case .off: repeatMode = .all
        case .all: repeatMode = .one
        case .one: repeatMode = .off
        }
    }
    
    public func setVolume(_ val: Float) {
        volume = max(0.0, min(1.0, val))
        isMuted = false
        player?.volume = volume
    }
    
    public func toggleMute() {
        isMuted.toggle()
        player?.volume = isMuted ? 0 : volume
    }
    
    private func addTimeObserver() {
        let interval = CMTime(seconds: 0.1, preferredTimescale: 600)
        timeObserverToken = player?.addPeriodicTimeObserver(forInterval: interval, queue: .main) { [weak self] time in
            guard let self = self else { return }
            let seconds = CMTimeGetSeconds(time)
            if !seconds.isNaN && !seconds.isInfinite {
                self.currentTime = seconds
                if let itemDuration = self.player?.currentItem?.duration {
                    let dSec = CMTimeGetSeconds(itemDuration)
                    if !dSec.isNaN && !dSec.isInfinite && dSec > 0 {
                        self.duration = dSec
                    }
                }
                self.updateLyricPosition(time: seconds)
            }
        }
    }
    
    private func removeTimeObserver() {
        if let token = timeObserverToken {
            player?.removeTimeObserver(token)
            timeObserverToken = nil
        }
    }
    
    private func updateLyricPosition(time: TimeInterval) {
        guard let lines = currentTrack?.lyrics?.lines, !lines.isEmpty else {
            activeLyricIndex = -1
            return
        }
        var foundIndex = -1
        for (idx, line) in lines.enumerated() {
            if line.timeSeconds <= time {
                foundIndex = idx
            } else {
                break
            }
        }
        if foundIndex != activeLyricIndex {
            activeLyricIndex = foundIndex
        }
    }
    
    private func handleTrackEnded() {
        if repeatMode == .one {
            seek(to: 0)
            play()
        } else {
            playNext()
        }
    }
    
    private func handleAudioInterruption(notification: Notification) {
        guard let userInfo = notification.userInfo,
              let typeValue = userInfo[AVAudioSessionInterruptionTypeKey] as? UInt,
              let type = AVAudioSession.InterruptionType(rawValue: typeValue) else {
            return
        }
        switch type {
        case .began:
            pause()
        case .ended:
            if let optionsValue = userInfo[AVAudioSessionInterruptionOptionKey] as? UInt {
                let options = AVAudioSession.InterruptionOptions(rawValue: optionsValue)
                if options.contains(.shouldResume) { play() }
            }
        @unknown default:
            break
        }
    }
    
    private func handleRouteChange(notification: Notification) {
        guard let userInfo = notification.userInfo,
              let reasonValue = userInfo[AVAudioSessionRouteChangeReasonKey] as? UInt,
              let reason = AVAudioSession.RouteChangeReason(rawValue: reasonValue) else {
            return
        }
        if reason == .oldDeviceUnavailable {
            pause()
        }
    }
    
    public func setupNowPlaying() {
        updateNowPlaying()
    }
    
    public func updateNowPlaying() {
        guard let track = currentTrack else {
            MPNowPlayingInfoCenter.default().nowPlayingInfo = nil
            return
        }
        
        var nowPlayingInfo: [String: Any] = [
            MPMediaItemPropertyTitle: track.title,
            MPMediaItemPropertyArtist: track.artistString,
            MPMediaItemPropertyPlaybackDuration: duration,
            MPNowPlayingInfoPropertyElapsedPlaybackTime: currentTime,
            MPNowPlayingInfoPropertyPlaybackRate: isPlaying ? 1.0 : 0.0
        ]
        
        if let album = track.album {
            nowPlayingInfo[MPMediaItemPropertyAlbumTitle] = album
        }
        
        if let url = URL(string: track.coverUrl) {
            URLSession.shared.dataTask(with: url) { data, _, _ in
                if let data = data, let image = UIImage(data: data) {
                    let artwork = MPMediaItemArtwork(boundsSize: image.size) { _ in image }
                    DispatchQueue.main.async {
                        var updated = MPNowPlayingInfoCenter.default().nowPlayingInfo ?? nowPlayingInfo
                        updated[MPMediaItemPropertyArtwork] = artwork
                        MPNowPlayingInfoCenter.default().nowPlayingInfo = updated
                    }
                }
            }.resume()
        }
        
        MPNowPlayingInfoCenter.default().nowPlayingInfo = nowPlayingInfo
    }
    
    private func updateNowPlayingPlaybackRate(_ rate: Float) {
        if var info = MPNowPlayingInfoCenter.default().nowPlayingInfo {
            info[MPNowPlayingInfoPropertyPlaybackRate] = rate
            info[MPNowPlayingInfoPropertyElapsedPlaybackTime] = currentTime
            MPNowPlayingInfoCenter.default().nowPlayingInfo = info
        }
    }
}
